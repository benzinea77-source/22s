-- ===================== SPINBOT =====================
local spinBAV, spinConn = nil, nil

local function stopSpinBot()
	if spinConn then spinConn:Disconnect() spinConn = nil end
	if spinBAV then spinBAV:Destroy() spinBAV = nil end
	local hrp = root()
	if hrp then
		for _, v in pairs(hrp:GetChildren()) do
			if v.Name == "SpinBAV" then v:Destroy() end
		end
		pcall(function()
			hrp.AssemblyAngularVelocity = Vector3.zero
			hrp.RotVelocity = Vector3.zero
		end)
	end
end

local function startSpinBot()
	stopSpinBot()
	local hrp = root()
	if not hrp then return end

	spinBAV = Instance.new("BodyAngularVelocity")
	spinBAV.Name = "SpinBAV"
	spinBAV.MaxTorque = Vector3.new(0, math.huge, 0)
	spinBAV.P = 1250
	spinBAV.AngularVelocity = Vector3.new(0, Values.SpinSpeed, 0)
	spinBAV.Parent = hrp

	spinConn = RunService.Heartbeat:Connect(function()
		if not Enabled.SpinBot then return end
		local h, r = humanoid(), root()
		if not h or not r then return end

		local state = h:GetState()
		if h.PlatformStand
			or state == Enum.HumanoidStateType.Physics
			or state == Enum.HumanoidStateType.Ragdoll
			or state == Enum.HumanoidStateType.FallingDown then
			if spinBAV then spinBAV.AngularVelocity = Vector3.new(0, 0, 0) end
			return
		end

		if Player:GetAttribute("Stealing") then
			if spinBAV then spinBAV.AngularVelocity = Vector3.new(0, 0, 0) end
			return
		end

		if spinBAV and spinBAV.Parent == r then
			spinBAV.AngularVelocity = Vector3.new(0, Values.SpinSpeed, 0)
		else
			if spinBAV then spinBAV:Destroy() end
			spinBAV = Instance.new("BodyAngularVelocity")
			spinBAV.Name = "SpinBAV"
			spinBAV.MaxTorque = Vector3.new(0, math.huge, 0)
			spinBAV.P = 1250
			spinBAV.AngularVelocity = Vector3.new(0, Values.SpinSpeed, 0)
			spinBAV.Parent = r
		end
	end)
end

-- ===================== SPEED (always active engine, modes via Q/R) =====================
local speedConn = nil
local SpeedModeLabel = nil -- set later from UI

local function activeSpeed()
	if SpeedMode.Lagger then
		return SpeedMode.Carry and Values.LaggerCarry or Values.LaggerNormal
	end
	return SpeedMode.Carry and Values.CarrySpeed or Values.NormalSpeed
end

local function refreshSpeedLabel()
	if not SpeedModeLabel then return end
	local mode
	if SpeedMode.Lagger then
		mode = SpeedMode.Carry and "LAGGER CARRY" or "LAGGER NORMAL"
	else
		mode = SpeedMode.Carry and "CARRY" or "NORMAL"
	end
	SpeedModeLabel.Text = "Running: " .. mode .. "  (" .. tostring(activeSpeed()) .. ")"
end

local function startSpeedEngine()
	if speedConn then return end
	speedConn = RunService.Heartbeat:Connect(function()
		local h, hrp = humanoid(), root()
		if not h or not hrp then return end

		local state = h:GetState()
		if h.PlatformStand
			or state == Enum.HumanoidStateType.Physics
			or state == Enum.HumanoidStateType.Ragdoll
			or state == Enum.HumanoidStateType.FallingDown then
			return
		end

		local md = h.MoveDirection
		if md.Magnitude <= 0.05 then return end

		local spd = activeSpeed()
		local curVel = hrp.AssemblyLinearVelocity
		hrp.AssemblyLinearVelocity = Vector3.new(md.X * spd, curVel.Y, md.Z * spd)
	end)
end

local function stopSpeedEngine()
	if speedConn then speedConn:Disconnect() speedConn = nil end
end

-- always run speed engine
task.spawn(function()
	task.wait(0.5)
	startSpeedEngine()
end)

	if Connections[name] then Connections[name]:Disconnect() Connections[name] = nil end

-- Anti Ragdoll
local function stopAntiRagdoll()
	disconnect("AntiRagdoll")
end
local function startAntiRagdoll()
	stopAntiRagdoll()
	Connections.AntiRagdoll = RunService.Heartbeat:Connect(function()
		if not Enabled.AntiRagdoll then return end
		local char = character()
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then return end
		local state = hum:GetState()
		if state == Enum.HumanoidStateType.Physics
			or state == Enum.HumanoidStateType.Ragdoll
			or state == Enum.HumanoidStateType.FallingDown then
			pcall(function()
				hum:ChangeState(Enum.HumanoidStateType.GettingUp)
				local rootPart = char:FindFirstChild("HumanoidRootPart")
				if rootPart then
					rootPart.Velocity = Vector3.zero
					rootPart.RotVelocity = Vector3.zero
					rootPart.AssemblyLinearVelocity = Vector3.zero
					rootPart.AssemblyAngularVelocity = Vector3.zero
				end
				for _, object in ipairs(char:GetDescendants()) do
					if object:IsA("Motor6D") then object.Enabled = true end
					if object:IsA("Constraint") then object.Enabled = true end
				end
				if workspace.CurrentCamera then workspace.CurrentCamera.CameraSubject = hum end
				hum.AutoRotate = true
				hum.PlatformStand = false
				hum.Sit = false
			end)
		end
	end)
end

-- Bat helpers
local function findBatTool()
	local c = character()
	if c then
		for _, t in ipairs(c:GetChildren()) do
			if t:IsA("Tool") then
				local n = t.Name:lower()
				if n:find("bat") or n:find("slap") then return t end
			end
		end
	end
	local bp = Player:FindFirstChild("Backpack")
	if bp then
		for _, t in ipairs(bp:GetChildren()) do
			if t:IsA("Tool") then
				local n = t.Name:lower()
				if n:find("bat") or n:find("slap") then return t end
			end
		end
	end
	return nil
end

local function getClosestTargetRoot()
	local hrp = root()
	if not hrp then return nil end
	local closest, minDist = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= Player and plr.Character then
			local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if tRoot and hum and hum.Health > 0 then
				local dist = (tRoot.Position - hrp.Position).Magnitude
				if dist < minDist then minDist = dist; closest = tRoot end
			end
		end
	end
	return closest
end

-- Bat Aimbot
local function stopBatAimbot()
	disconnect("BatAimbot")
	local h0, r0 = humanoid(), root()
	if r0 then
		r0.AssemblyLinearVelocity = Vector3.zero
		r0.AssemblyAngularVelocity = Vector3.zero
	end
	if h0 then h0.AutoRotate = true end
end

local function startBatAimbot()
	stopBatAimbot()
	if Enabled.TPBat then
		Enabled.TPBat = false
		if VisualSetters.TPBat then VisualSetters.TPBat(false, true) end
		stopTPBat()
	end
	local h0 = humanoid()
	if h0 then h0.AutoRotate = false end
	local swingCd = false
	Connections.BatAimbot = RunService.RenderStepped:Connect(function()
		if not Enabled.BatAimbot then return end
		local char = character()
		if not char then return end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local hum = char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum then return end
		local bat = char:FindFirstChildOfClass("Tool") or findBatTool()
		if bat and bat.Parent ~= char then pcall(function() hum:EquipTool(bat) end) end
		local target = getClosestTargetRoot()
		if not target then return end
		local targetVel = target.AssemblyLinearVelocity
		local myPos = hrp.Position
		local targetPos = target.Position
		local predictPos = targetPos + targetVel * 0.14 + target.CFrame.LookVector * 0.3
		local direction = predictPos - myPos
		if direction.Magnitude < 0.01 then return end
		local flatDir = Vector3.new(direction.X, 0, direction.Z)
		if flatDir.Magnitude < 0.01 then return end
		flatDir = flatDir.Unit
		local chaseSpeed = Values.AimbotSpeed or 50
		local desiredHeight = targetPos.Y + 3.7
		local yVel = (desiredHeight - myPos.Y) * 19.5 + targetVel.Y * 0.8
		if hum.FloorMaterial ~= Enum.Material.Air then yVel = math.max(yVel, 13) end
		yVel = math.clamp(yVel, -70, 110)
		hrp.AssemblyLinearVelocity = hrp.AssemblyLinearVelocity:Lerp(
			Vector3.new(flatDir.X * chaseSpeed, yVel, flatDir.Z * chaseSpeed), 0.8
		)
		local predictTime = math.clamp(targetVel.Magnitude / 150, 0.05, 0.2)
		local predictedPos = targetPos + targetVel * predictTime
		local toPredict = predictedPos - myPos
		if toPredict.Magnitude > 0.1 then
			local goalCF = CFrame.lookAt(myPos, predictedPos)
			local diffCF = hrp.CFrame:Inverse() * goalCF
			local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
			hrp.AssemblyAngularVelocity = hrp.CFrame:VectorToWorldSpace(Vector3.new(
				math.clamp(rx, -2.5, 2.5) * 42,
				math.clamp(ry, -2.5, 2.5) * 42,
				math.clamp(rz, -2.5, 2.5) * 42
			))
		end
		if Enabled.AutoSwing and bat and not swingCd then
			swingCd = true
			pcall(function() bat:Activate() end)
			task.delay(0.08, function() swingCd = false end)
		end
	end)
end

-- TP Bat (AntiDesync simplified orbit)
local function stopTPBat()
	disconnect("TPBat")
	local hrp = root()
	if hrp and sethiddenproperty then
		pcall(function() sethiddenproperty(hrp, "PhysicsRepRootPart", hrp) end)
	end
end

local function startTPBat()
	stopTPBat()
	if Enabled.BatAimbot then
		Enabled.BatAimbot = false
		if VisualSetters.BatAimbot then VisualSetters.BatAimbot(false, true) end
		stopBatAimbot()
	end
	local hittingCooldown = false
	local spinAngle = 0
	local function tryHit()
		if hittingCooldown then return end
		hittingCooldown = true
		pcall(function()
			local bat = findBatTool()
			local char = character()
			local hum = humanoid()
			if bat and char and hum then
				if bat.Parent ~= char then pcall(function() hum:EquipTool(bat) end) end
				for _ = 1, 3 do
					pcall(function() bat:Activate() end)
				end
			end
		end)
		task.delay(0.045, function() hittingCooldown = false end)
	end
	Connections.TPBat = RunService.Heartbeat:Connect(function(dt)
		if not Enabled.TPBat then return end
		local hrp = root()
		if not hrp then return end
		local target = getClosestTargetRoot()
		if not target then return end
		if sethiddenproperty then
			pcall(function() sethiddenproperty(hrp, "PhysicsRepRootPart", target) end)
		end
		dt = (type(dt) == "number" and dt > 0 and dt < 0.1) and dt or (1/60)
		local center = target.Position + Vector3.new(0, 0.9, 0)
		local tVel = target.AssemblyLinearVelocity
		local predicted = center + Vector3.new(tVel.X, 0, tVel.Z) * 0.06
		spinAngle = spinAngle + 48 * dt * 2.2
		local spin = spinAngle
		local face = predicted + Vector3.new(math.sin(spin) * 0.4, 0, math.cos(spin) * 0.4)
		hrp.CFrame = CFrame.new(predicted + Vector3.new(0, 0.15, 0), face)
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
		tryHit()
	end)
end

-- Instant Reset (V1 hipheight style)
local function runInstantReset()
	local char = character()
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	local original = hum.HipHeight
	local attempts = 0
	task.spawn(function()
		while char and char.Parent and hum and hum.Health > 0 and Player.Character == char and attempts < 40 do
			pcall(function()
				hum.HipHeight = 1e30
				hum.AutoRotate = true
				for _, part in ipairs(char:GetChildren()) do
					if part:IsA("BasePart") then part.CanCollide = false end
				end
			end)
			if not char.Parent or hum.Health <= 0 or Player.Character ~= char then break end
			attempts += 1
			task.wait(0.05)
		end
		if char and char.Parent and hum and hum.Health > 0 then
			pcall(function() hum.Health = 0 end)
		end
	end)
end


-- Bat Counter (activate bat when ragdolled)
local _batCounterDebounce = false
local function stopBatCounter()
	disconnect("BatCounter")
	Connections.BatCounterToken = nil
	_batCounterDebounce = false
end
local function startBatCounter()
	stopBatCounter()
	local token = {}
	Connections.BatCounterToken = token
	task.spawn(function()
		while Connections.BatCounterToken == token and Enabled.BatCounter do
			if not _batCounterDebounce then
				local h = humanoid()
				if h then
					local st = h:GetState()
					local rag = st == Enum.HumanoidStateType.Physics
						or st == Enum.HumanoidStateType.Ragdoll
						or st == Enum.HumanoidStateType.FallingDown
					if rag then
						_batCounterDebounce = true
						local bat = findBatTool()
						local c = character()
						if bat and c then
							if bat.Parent ~= c then pcall(function() h:EquipTool(bat) end) end
							pcall(function() bat:Activate() end)
							task.wait(0.15)
							pcall(function() bat:Activate() end)
						end
						task.wait(0.45)
						_batCounterDebounce = false
					end
				end
			end
			task.wait(0.12)
		end
	end)
end

-- Medusa Counter (activate medusa tool when stone/anchor trap)
local _medusaAnchorConns = {}
local _medusaDebounce, _medusaLastUsed = false, 0
local function _findMedusaTool()
	local c = character()
	if not c then return nil end
	for _, t in ipairs(c:GetChildren()) do
		if t:IsA("Tool") then
			local n = t.Name:lower()
			if n:find("medusa") or n:find("head") or n:find("stone") then return t end
		end
	end
	local bp = Player:FindFirstChildOfClass("Backpack")
	if bp then
		for _, t in ipairs(bp:GetChildren()) do
			if t:IsA("Tool") then
				local n = t.Name:lower()
				if n:find("medusa") or n:find("head") or n:find("stone") then return t end
			end
		end
	end
	return nil
end
local function _useMedusaCounter()
	if _medusaDebounce or tick() - _medusaLastUsed < 0.5 then return end
	local c = character(); if not c then return end
	_medusaDebounce = true
	local med = _findMedusaTool()
	if not med then _medusaDebounce = false; return end
	if med.Parent ~= c then
		local h = humanoid(); if h then pcall(function() h:EquipTool(med) end) end
	end
	pcall(function() med:Activate() end)
	_medusaLastUsed = tick()
	_medusaDebounce = false
end
local function stopMedusaCounter()
	for _, c in pairs(_medusaAnchorConns) do pcall(function() c:Disconnect() end) end
	table.clear(_medusaAnchorConns)
	disconnect("Medusa")
end
local function startMedusaCounter()
	stopMedusaCounter()
	local function watchPart(part)
		if not part:IsA("BasePart") then return end
		table.insert(_medusaAnchorConns, part:GetPropertyChangedSignal("Anchored"):Connect(function()

-- ── helpers ──────────────────────────────────────────────────
local function isValidImage(d)
    if type(d) ~= "string" or #d <= 64 then return false end
    if d:sub(1, 4) == "\137PNG" then return true end
    if d:byte(1) == 255 and d:byte(2) == 216 then return true end
    return false
end

local function httpGet(url)
    local req = request or http_request or (syn and syn.request) or (http and http.request) or (getgenv and getgenv().request)
    if type(req) == "function" then
        local ok, r = pcall(req, { Url = url, Method = "GET" })
        if ok then
            if type(r) == "table" then
                local b = r.Body or r.body or r.Content or r.content
                if type(b) == "string" and #b > 0 then return true, b end
            elseif type(r) == "string" and #r > 0 then
                return true, r
            end
        end
    end
    return pcall(game.HttpGet, game, url)
end

local function downloadAsset(filePath, url)
    if not getcustomasset then return nil end
    if isfile and isfile(filePath) then
        local ok, data = pcall(readfile, filePath)
        if isValidImage(ok and data or nil) then
            local ok2, a = pcall(getcustomasset, filePath)
            if ok2 and a and a ~= "" then return a end
        end
    end
    if url then
        local ok, body = httpGet(url)
        if ok and isValidImage(body) then
            if writefile then pcall(writefile, filePath, body) end
            local ok2, a = pcall(getcustomasset, filePath)
            if ok2 and a and a ~= "" then return a end
        end
    end
    return nil
end

local function saveWearables(character)
    skinState.savedWearables = {}
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Accessory") or child:IsA("Shirt") or child:IsA("Pants")
           or child:IsA("ShirtGraphic") or child:IsA("CharacterMesh") then
            local ok, clone = pcall(function() local c = child:Clone(); c.Parent = nil; return c end)
            if ok and clone then table.insert(skinState.savedWearables, clone) end
        end
    end
    for _, desc in ipairs(character:GetDescendants()) do
        if (desc:IsA("Decal") or desc:IsA("Texture")) and desc.Parent and desc.Parent.Name == "Head" then
            local ok, clone = pcall(function() local c = desc:Clone(); c.Parent = nil; return c end)
            if ok and clone then table.insert(skinState.savedWearables, { node = clone }) end
        end
    end
end

local function restoreWearables(character)
    for _, w in ipairs(skinState.savedWearables) do
        pcall(function()
            if type(w) == "table" and w.node then
                local head = character:FindFirstChild("Head")
                if head then w.node.Parent = head end
            elseif w and w.Parent == nil then
                w.Parent = character
            end
        end)
    end
    skinState.savedWearables = {}
end

end

local function restoreOriginalBody()
    for object, saved in pairs(state.hidden) do
        if object and object.Parent then
            pcall(function()
                if type(saved) == "table" and saved.basePart then
                    object.LocalTransparencyModifier = saved.localTransparency
                    object.Transparency = saved.transparency
                elseif type(saved) == "table" and saved.properties then
                    for p, v in pairs(saved.properties) do object[p] = v end
                elseif type(saved) == "table" and saved.property then
                    object[saved.property] = saved.value
                elseif object:IsA("BasePart") then
                    object.LocalTransparencyModifier = saved
                elseif object:IsA("Decal") or object:IsA("Texture") then
                    object.Transparency = saved
                end
            end)
        end
    end
    state.hidden = {}
end

local function clearVisuals()
    stopHeadless()
    if state.wearableConn then
        state.wearableConn:Disconnect()
        state.wearableConn = nil
    end
    for _, object in ipairs(state.visuals) do
        if object and object.Parent then pcall(function() object:Destroy() end) end
    end
    state.visuals = {}
    restoreOriginalBody()
end
local function belongsToLocalVisual(object)
    for _, visual in ipairs(state.visuals) do
        if object == visual or object:IsDescendantOf(visual) then return true end
    end
    return false
end
local function hideOriginalBody(character)
    restoreOriginalBody()
    for _, object in ipairs(character:GetDescendants()) do
        if not belongsToLocalVisual(object) then
            if object:IsA("BasePart") then
                state.hidden[object] = object.LocalTransparencyModifier
                object.LocalTransparencyModifier = 1
            elseif object:IsA("Decal") or object:IsA("Texture") then
                state.hidden[object] = object.Transparency
                object.Transparency = 1
            end
        end
    end
end
local function hideOriginalHead(character)
    local head = character:FindFirstChild("Head")
    if not head or not head:IsA("BasePart") then return end
    if state.hidden[head] == nil then
        state.hidden[head] = {basePart=true, localTransparency=head.LocalTransparencyModifier, transparency=head.Transparency}
    elseif type(state.hidden[head]) ~= "table" then
        state.hidden[head] = {basePart=true, localTransparency=state.hidden[head], transparency=head.Transparency}
    end
    head.LocalTransparencyModifier = 1
    head.Transparency = 1
    for _, object in ipairs(head:GetDescendants()) do
        if (object:IsA("Decal") or object:IsA("Texture")) and state.hidden[object] == nil then
            state.hidden[object] = object.Transparency
            object.Transparency = 1
        end
    end
end

stopHeadless = function()
    if state.headlessConn then pcall(state.headlessConn.Disconnect, state.headlessConn); state.headlessConn = nil end
    if state.headlessLoop then pcall(task.cancel, state.headlessLoop); state.headlessLoop = nil end
    if state.statsLoop then pcall(task.cancel, state.statsLoop); state.statsLoop = nil end
end
enforceHeadless = function(character, option)
    stopHeadless()
    if not option or not character or not (option.headless or option.hideLeftLeg or option.hideRightLeg) then return end
    if option.headless then hideOriginalHead(character) end
    if option.hideLeftLeg then hideLeftLeg(character) end
    if option.hideRightLeg then hideRightLeg(character) end
    state.headlessConn = character.ChildAdded:Connect(function(child)
        if child:IsA("BasePart") and (option.hideLeftLeg or option.hideRightLeg) then
            task.defer(function()
                if state.enabled and LP.Character == character then
                    local active = OUTFITS[state.selected]
                    if active.hideLeftLeg then hideLeftLeg(character) end
                    if active.hideRightLeg then hideRightLeg(character) end
                end
            end)
        end
        if child.Name == "Head" and child:IsA("BasePart") and OUTFITS[state.selected].headless then
            task.defer(function()
                if state.enabled and LP.Character == character and OUTFITS[state.selected].headless then
                    pcall(hideOriginalHead, character)
                end
            end)
        end
    end)
    state.headlessLoop = task.spawn(function()
        while state.enabled and LP.Character == character do
            local opt = OUTFITS[state.selected]
            if opt and opt.headless then
                local head = character:FindFirstChild("Head")
                if head and head:IsA("BasePart") and head.LocalTransparencyModifier < 1 then
                    pcall(hideOriginalHead, character)
                end
            end
            if opt and opt.hideLeftLeg then pcall(hideLeftLeg, character) end
            if opt and opt.hideRightLeg then pcall(hideRightLeg, character) end
            task.wait(0.6)
        end
    end)
end
local function enforceStats(character)
    if state.statsLoop then pcall(task.cancel, state.statsLoop); state.statsLoop = nil end
    state.statsLoop = task.spawn(function()
        while state.enabled and LP.Character == character do
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                if state.savedWalkSpeed ~= nil and humanoid.WalkSpeed ~= state.savedWalkSpeed then
                    humanoid.WalkSpeed = state.savedWalkSpeed
                end
                if state.savedJumpPower ~= nil and humanoid.JumpPower ~= state.savedJumpPower then
                    humanoid.JumpPower = state.savedJumpPower
                end
            end
            task.wait(0.4)
        end
    end)
end
local function hideOriginalWearables(character, watch)
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") then
            if not child:GetAttribute("VX7LocalOnly") and state.hidden[child] == nil then
                local prop = child:IsA("Shirt") and "ShirtTemplate"
                    or child:IsA("Pants") and "PantsTemplate" or "Graphic"
                state.hidden[child] = {property=prop, value=child[prop]}
                child[prop] = ""
            end
        end
    end
    for _, child in ipairs(character:GetDescendants()) do
        if (child:IsA("Accessory") or child:IsA("Accoutrement")) and not child:GetAttribute("VX7LocalOnly") then
            local objects = child:GetDescendants()
            if child:IsA("BasePart") then table.insert(objects, child) end
            for _, object in ipairs(objects) do
                if object:IsA("BasePart") then
                    local saved = state.hidden[object]
                    if saved == nil then
                        state.hidden[object] = {basePart=true, localTransparency=object.LocalTransparencyModifier, transparency=object.Transparency}
                    elseif type(saved) ~= "table" then
                        state.hidden[object] = {basePart=true, localTransparency=saved, transparency=object.Transparency}
                    end
                    object.LocalTransparencyModifier = 1
                    object.Transparency = 1
                elseif (object:IsA("Decal") or object:IsA("Texture")) and state.hidden[object] == nil then
                    state.hidden[object] = object.Transparency
                    object.Transparency = 1
                elseif (object:IsA("ParticleEmitter") or object:IsA("Trail") or object:IsA("Beam")) and state.hidden[object] == nil then
                    state.hidden[object] = {property="Enabled", value=object.Enabled}
                    object.Enabled = false
                end
            end
        end
    end
    if watch and not state.wearableConn then
        state.wearableConn = character.DescendantAdded:Connect(function(object)
            if not state.enabled or not OUTFITS[state.selected].clearOriginalWearables then return end
            if object:IsA("Accessory") or object:IsA("Accoutrement")
               or object:FindFirstAncestorWhichIsA("Accessory")
               or object:FindFirstAncestorWhichIsA("Accoutrement") then
                task.defer(function()
                    if state.enabled and character.Parent then hideOriginalWearables(character, false) end
                end)
            end
        end)
    end
end
local function loadAssetObjects(assetId)
    local ok, inserted = pcall(function() return InsertService:LoadAsset(assetId) end)
    if ok and inserted then return {inserted}, "InsertService" end
    local ok2, objects = pcall(function() return game:GetObjects("rbxassetid://" .. assetId) end)
    if ok2 and objects and #objects > 0 then return objects, "GetObjects" end
    return nil, tostring(inserted or objects or "unknown")
end
local faceTextureCache = {}
local function resolveFaceTexture(faceId)
    if faceTextureCache[faceId] then return faceTextureCache[faceId] end
    local objects = loadAssetObjects(faceId)
    if objects then
        local texture
        for _, root in ipairs(objects) do
            local candidates = {root}
            for _, o in ipairs(root:GetDescendants()) do table.insert(candidates, o) end
            for _, o in ipairs(candidates) do
                if o:IsA("Decal") or o:IsA("Texture") then
                    texture = o.Texture
                    if texture and texture ~= "" then break end
                end
            end
            pcall(function() root:Destroy() end)
            if texture and texture ~= "" then break end
        end
        if texture and texture ~= "" then
            faceTextureCache[faceId] = texture
            return texture
        end
    end
    return "rbxassetid://" .. tostring(faceId)
end
local function applyLocalFace(character, faceId)
    local head = character:FindFirstChild("Head")
    if not head then return false end
    local face = head:FindFirstChild("face") or head:FindFirstChildWhichIsA("Decal")
    if face and not face:IsA("Decal") then face = nil end
    if face then
        if state.hidden[face] == nil then
            state.hidden[face] = {properties={Texture=face.Texture, Transparency=face.Transparency}}
        end
    else
        face = Instance.new("Decal")
        face.Name = "face"
        face.Face = Enum.NormalId.Front
        face:SetAttribute("VX7LocalOnly", true)
        face.Parent = head
        table.insert(state.visuals, face)
    end
    face.Texture = resolveFaceTexture(faceId)
    face.Transparency = 0
    return true
end
local function applyLocalFaceStable(character, faceId)
    pcall(applyLocalFace, character, faceId)
    for _, d in ipairs({0.2, 0.8}) do
        task.delay(d, function()
            local opt = OUTFITS[state.selected]
            if state.enabled and LP.Character == character and opt and opt.face == faceId then
                pcall(applyLocalFace, character, faceId)
            end
        end)
    end
end

local function sanitizeVisual(container)
    local objects = container:GetDescendants()
    if container:IsA("BasePart") then table.insert(objects, container) end
    for _, o in ipairs(objects) do
        if o:IsA("LuaSourceContainer") then o:Destroy()
        elseif o:IsA("BasePart") then
            o.Anchored = false; o.CanCollide = false; o.CanTouch = false
            o.CanQuery = false; o.Massless = true; o.CastShadow = false
        elseif o:IsA("JointInstance") and o.Name == "AccessoryWeld" then o:Destroy() end
    end
end
local function weldPart(part, target)
    local weld = Instance.new("WeldConstraint")
    weld.Name = "VX7LocalVisualWeld"
    weld.Part0 = target; weld.Part1 = part; weld.Parent = part
end
local function countStackedAccessories(character, attachmentName)
    local count = 0
    for _, visual in ipairs(state.visuals) do
        if visual:IsA("Accessory") then
            local h = visual:FindFirstChild("Handle") or visual:FindFirstChildWhichIsA("BasePart", true)
            if h then
                for _, child in ipairs(h:GetChildren()) do
                    if child:IsA("Attachment") and child.Name == attachmentName then
                        count = count + 1
                        break
                    end
                end
            end
        end
    end
    return count
end
local function attachAccessoryLocal(accessory, character, rootPart, assetId)
    local handle = accessory:FindFirstChild("Handle") or accessory:FindFirstChildWhichIsA("BasePart", true)
    if not handle then return false end
    sanitizeVisual(accessory)
    accessory.Name = "VX7_LocalOutfit_" .. tostring(assetId)
    accessory:SetAttribute("VX7AssetId", assetId)
    accessory:SetAttribute("VX7LocalOnly", true)
    local opt = OUTFITS[state.selected]
    local slot = opt and opt.accessorySlots and opt.accessorySlots[assetId]
    if slot then pcall(function() accessory.AccessoryType = slot end) end
    accessory.Parent = character
    local sourceAttachment
    for _, child in ipairs(handle:GetChildren()) do
        if child:IsA("Attachment") and findCharacterAttachment(character, child.Name) then
            sourceAttachment = child; break
        end
    end
    local targetAttachment = sourceAttachment and findCharacterAttachment(character, sourceAttachment.Name)
    local targetPart = targetAttachment and targetAttachment.Parent
    if targetAttachment and targetPart and targetPart:IsA("BasePart") then
        handle.CFrame = targetAttachment.WorldCFrame * sourceAttachment.CFrame:Inverse()
        local stacked = countStackedAccessories(character, sourceAttachment.Name)
        if stacked > 0 then handle.CFrame = handle.CFrame * CFrame.new(0, stacked * 0.15, 0) end
    else
        targetPart = character:FindFirstChild("LowerTorso") or character:FindFirstChild("Torso") or rootPart
        if not targetPart then accessory:Destroy(); return false end
        handle.CFrame = rootPart.CFrame
    end
    weldPart(handle, targetPart)
    table.insert(state.visuals, accessory)
    return true
end
local function attachLooseVisual(rootObject, character, rootPart, assetId, textureId, targetPart)
    local clone = rootObject:Clone()
    sanitizeVisual(clone)
    clone.Name = "VX7_LocalOutfit_" .. tostring(assetId)
    clone:SetAttribute("VX7AssetId", assetId)
    clone.Parent = character
    local parts = {}
    if clone:IsA("BasePart") then table.insert(parts, clone) end
    for _, o in ipairs(clone:GetDescendants()) do
        if o:IsA("BasePart") then table.insert(parts, o) end
    end
    if #parts == 0 then clone:Destroy(); return false end
    local anchor = targetPart or rootPart
    if textureId then
        for _, part in ipairs(parts) do
            if part:IsA("MeshPart") then
                pcall(function() part.TextureID = "rbxassetid://" .. tostring(textureId) end)
            end
        end
    end
    local pivot = clone:IsA("Model") and clone:GetPivot() or parts[1].CFrame
    for _, part in ipairs(parts) do
        local offset = pivot:ToObjectSpace(part.CFrame)
        part.CFrame = anchor.CFrame * offset
        weldPart(part, anchor)
    end
    table.insert(state.visuals, clone)
    return true
end

local function attachBodyPartsLocal(rootObject, character, assetId, textureId)
    local holder = Instance.new("Model")
    holder.Name = "VX7_LocalBody_" .. tostring(assetId)
    holder:SetAttribute("VX7AssetId", assetId)
    holder:SetAttribute("VX7LocalOnly", true)
    holder.Parent = character
    local bodyColors = rootObject:FindFirstChildOfClass("BodyColors")
    local colorProperties = {
        Head="HeadColor", UpperTorso="TorsoColor", LowerTorso="TorsoColor", Torso="TorsoColor",
        LeftUpperArm="LeftArmColor", LeftLowerArm="LeftArmColor", LeftHand="LeftArmColor", ["Left Arm"]="LeftArmColor",
        RightUpperArm="RightArmColor", RightLowerArm="RightArmColor", RightHand="RightArmColor", ["Right Arm"]="RightArmColor",
        LeftUpperLeg="LeftLegColor", LeftLowerLeg="LeftLegColor", LeftFoot="LeftLegColor", ["Left Leg"]="LeftLegColor",
        RightUpperLeg="RightLegColor", RightLowerLeg="RightLegColor", RightFoot="RightLegColor", ["Right Leg"]="RightLegColor",
    }
    local added = 0
    local candidates = {}
    if rootObject:IsA("BasePart") then table.insert(candidates, rootObject) end
    for _, o in ipairs(rootObject:GetDescendants()) do
        if o:IsA("BasePart") and not o:FindFirstAncestorWhichIsA("Accessory") then
            table.insert(candidates, o)
        end
    end
    for _, source in ipairs(candidates) do
        local partName = source.Name
        if BODY_PART_NAMES[partName] then
            local target = character:FindFirstChild(partName)
            if not target then
                local alias = PART_NAME_ALIASES[partName]
                if alias then target = character:FindFirstChild(alias) end
            end
            if target and target:IsA("BasePart") then
                local clone = source:Clone()
                for _, child in ipairs(clone:GetDescendants()) do
                    if child:IsA("JointInstance") or child:IsA("Constraint") or child:IsA("LuaSourceContainer") then
                        child:Destroy()
                    end
                end
                sanitizeVisual(clone)
                clone.Name = source.Name
                if clone:IsA("MeshPart") and textureId then
                    pcall(function() clone.TextureID = "rbxassetid://" .. tostring(textureId) end)
                end
                if bodyColors and colorProperties[source.Name] then
                    pcall(function() clone.Color = bodyColors[colorProperties[source.Name]].Color end)
                end
                clone.CFrame = target.CFrame
                clone.Parent = holder
                weldPart(clone, target)
                added = added + 1
            end
        end
    end
    if added == 0 then holder:Destroy(); return 0 end
    table.insert(state.visuals, holder)
    return added
end

    "BackAccessory","FaceAccessory","FrontAccessory","HairAccessory","HatAccessory",
    "NeckAccessory","ShouldersAccessory","WaistAccessory","TShirtAccessory",
    "ShirtAccessory","PantsAccessory","JacketAccessory","SweaterAccessory",

local function createDirectDescription(option, humanoid)
    local directOutfitId = option.outfitId
    local complete
    if directOutfitId then
        local ok, desc = pcall(Players.GetHumanoidDescriptionFromOutfitId, Players, directOutfitId)
        if not ok or not desc then return nil end
        complete = desc
        if not option.items and not option.face and not option.head then return desc end
    end
    local source = complete or state.originalDescription
    if not source then
        local ok, applied = pcall(function() return humanoid:GetAppliedDescription() end)
        if not ok or not applied then return nil end
        source = applied
    end
    local description = source:Clone()
    if complete then complete:Destroy() end
    if not directOutfitId then
        description.Shirt = 0
        description.Pants = 0
        description.GraphicTShirt = 0
        for _, p in ipairs(accessoryDescriptionProperties) do
            pcall(function() description[p] = "" end)
        end
        pcall(function() description:SetAccessories({}, true) end)
    end
    local accessories = {}
    if directOutfitId then pcall(function() accessories = description:GetAccessories(true) end) end
    local allItems = {}
    for _, id in ipairs(option.bodyItems or {}) do table.insert(allItems, id) end
    for _, id in ipairs(option.items or {}) do table.insert(allItems, id) end
    for _, id in ipairs(allItems) do
        local at = getAssetTypeId(id)
        if not at then description:Destroy(); return nil end
        local bodyProp = bodyPropertyByType[at]
        local accType = accessoryTypeByAssetType[at]
        if bodyProp then
            description[bodyProp] = id
        elseif at == 11 then description.Shirt = id
        elseif at == 12 then description.Pants = id
        elseif at == 1 then description.GraphicTShirt = id
        elseif at == 18 then description.Face = id
        elseif accType then
            local acc = {AssetId=id, AccessoryType=accType, Order=#accessories + 1}
            if at >= 64 and at <= 72 then
                acc.IsLayered = true
                acc.Puffiness = option.layeredPuffiness or 0
            end
            table.insert(accessories, acc)
        else
            description:Destroy(); return nil
        end
    end
    if option.face then
        local ft = getAssetTypeId(option.face)
        local fbp = ft and bodyPropertyByType[ft]
        local fat = ft and accessoryTypeByAssetType[ft]
        if option.faceMode == "decal" then
            if ft == 18 then description.Face = option.face end
        elseif fbp then description[fbp] = option.face
        elseif fat then
            table.insert(accessories, {AssetId=option.face, AccessoryType=fat, Order=#accessories + 1})
        elseif ft == 18 then description.Face = option.face
        end
    end
    if option.head then description.Head = option.head end
    for prop, val in pairs(option.scales or {}) do pcall(function() description[prop] = val end) end
    if option.skinColor then
        for _, p in ipairs({"HeadColor","LeftArmColor","RightArmColor","LeftLegColor","RightLegColor","TorsoColor"}) do
            description[p] = option.skinColor
        end
    end
    if #accessories > 0 then
        local ok = pcall(function() description:SetAccessories(accessories, true) end)
        if not ok then description:Destroy(); return nil end
    end
    return description
end
local function applyOutfit()
    local character, humanoid, rootPart = currentCharacter()
    if not character or not humanoid or not rootPart then return false end
    if state.descriptionCharacter ~= character or not state.originalDescription then
        state.descriptionCharacter = character
        state.originalDescription = nil
        local ok, applied = pcall(function() return humanoid:GetAppliedDescription() end)
        if ok and applied then state.originalDescription = applied:Clone() end
    end
    clearVisuals()
    local option = OUTFITS[state.selected]
    local assetId = option.id or option.outfitId
    state.skipDescriptionRestore = option.preserveRig == true

    if (option.bodyItems or option.items or option.outfitId) and not option.forceBodyParts and not option.localOnly then
        local description = createDirectDescription(option, humanoid)
        if description then
            local savedHead = character:FindFirstChild("Head")
            local savedHealth = humanoid.Health
            state.savedWalkSpeed = humanoid.WalkSpeed
            state.savedJumpPower = humanoid.JumpPower
            local savedCFrame = rootPart.CFrame
            local savedVel = rootPart.AssemblyLinearVelocity
            local ok = pcall(function() humanoid:ApplyDescription(description) end)
            if not ok then ok = pcall(function() humanoid:ApplyDescriptionReset(description) end) end
            description:Destroy()
            if ok then
                local curHum = character:FindFirstChildOfClass("Humanoid")
                local curRoot = character:FindFirstChild("HumanoidRootPart")
                if curHum and savedHealth > 0 then curHum.Health = math.min(savedHealth, curHum.MaxHealth) end
                if curRoot then
                    curRoot.CFrame = savedCFrame
                    curRoot.AssemblyLinearVelocity = savedVel
                    curRoot.AssemblyAngularVelocity = Vector3.zero
                end
                if option.face then
                    local ft = getAssetTypeId(option.face)
                    if option.faceMode == "decal" or not ft or ft == 18 then
                        applyLocalFaceStable(character, option.face)
                    end
                end
                state.enabled = true
                enforceStats(character)
                enforceHeadless(character, option)
                if option.hideLeftLeg then hideLeftLeg(character) end
                if option.hideRightLeg then hideRightLeg(character) end
                if option.head and character:FindFirstChild("Head") == savedHead then
                    hideOriginalHead(character)
                end
                return true
            end
        end
    end

    -- fallback: local visual attachment
    local assetIds = {}
    for _, id in ipairs(option.items or {}) do table.insert(assetIds, id) end
    if option.head then table.insert(assetIds, option.head) end
    if #assetIds == 0 and assetId and not option.outfitId then table.insert(assetIds, assetId) end
    local fallbackFaceType = option.face and getAssetTypeId(option.face)
    local faceIsAccessory = option.faceMode ~= "decal" and fallbackFaceType and accessoryTypeByAssetType[fallbackFaceType] ~= nil
    if faceIsAccessory then table.insert(assetIds, option.face) end
    local attached, expected, lastErr = 0, 0, nil
    expected = expected + (option.face and not faceIsAccessory and 1 or 0)
    for _, curId in ipairs(assetIds) do
        local objects = loadAssetObjects(curId)
        if objects then
            for _, loadedRoot in ipairs(objects) do
                local rootAttached, bodyPartsAttached = false, false
                local classics = {}
                if loadedRoot:IsA("Shirt") or loadedRoot:IsA("Pants") or loadedRoot:IsA("ShirtGraphic") then
                    table.insert(classics, loadedRoot)
                end
                for _, o in ipairs(loadedRoot:GetDescendants()) do
                    if o:IsA("Shirt") or o:IsA("Pants") or o:IsA("ShirtGraphic") then
                        table.insert(classics, o)
                    end
                end
                local accessories = {}
                if loadedRoot:IsA("Accessory") then
                    table.insert(accessories, loadedRoot)
                else
                    for _, o in ipairs(loadedRoot:GetDescendants()) do
                        if o:IsA("Accessory") then table.insert(accessories, o:Clone()) end
                    end
                end
                local at = getAssetTypeId(curId)
                local isBodyPartAsset = at ~= nil and bodyPropertyByType[at] ~= nil
                local bodyCandidates = 0
                local wantsBody = not loadedRoot:IsA("Accessory")
                    and (option.hideOriginal or option.hideOriginalHead or curId == option.head
                         or (option.forceBodyParts and isBodyPartAsset))
                if wantsBody then
                    local candidates = {}
                    if loadedRoot:IsA("BasePart") then table.insert(candidates, loadedRoot) end
                    for _, o in ipairs(loadedRoot:GetDescendants()) do
                        if o:IsA("BasePart") and not o:FindFirstAncestorWhichIsA("Accessory") then
                            table.insert(candidates, o)
                        end
                    end
                    for _, o in ipairs(candidates) do
                        if BODY_PART_NAMES[o.Name] or PART_NAME_ALIASES[o.Name] then bodyCandidates = bodyCandidates + 1 end
                    end
                end
                expected = expected + #classics + #accessories + bodyCandidates
                if #accessories == 0 and #classics == 0 and bodyCandidates == 0 then expected = expected + 1 end
                if #classics > 0 then
                    local ok, r = pcall(attachClassicClothingLocal, loadedRoot, character, curId)
                    if ok and r and r > 0 then attached = attached + r; rootAttached = true end
                end
                if wantsBody and bodyCandidates > 0 then
                    local ok, r = pcall(attachBodyPartsLocal, loadedRoot, character, curId, option.headTexture)
                    if ok and r and r > 0 then
                        attached = attached + r; rootAttached = true; bodyPartsAttached = true
                    end
                end
                for _, acc in ipairs(accessories) do
                    local ok, r = pcall(attachAccessoryLocal, acc, character, rootPart, curId)
                    if ok and r then attached = attached + 1; rootAttached = true
                    else pcall(function() acc:Destroy() end) end
                end
                if not rootAttached and not bodyPartsAttached and #accessories == 0 and #classics == 0 then
                    local headAnchor = character:FindFirstChild("Head")
                    local anchor = rootPart
                    if curId == option.head then anchor = headAnchor
                    elseif isBodyPartAsset then
                        local anchorNames = bodyPartAnchorNames[at]
                        if anchorNames then
                            for _, name in ipairs(anchorNames) do
                                local t = character:FindFirstChild(name)
                                if t and t:IsA("BasePart") then anchor = t; break end
                            end
                        end
                    end
                    local ok, r = pcall(attachLooseVisual, loadedRoot, character, rootPart, curId, option.headTexture, anchor)
                    if ok and r then attached = attached + 1 end
                end
                if loadedRoot.Parent == nil and not table.find(state.visuals, loadedRoot) then
                    pcall(function() loadedRoot:Destroy() end)
                end
            end
        else
            lastErr = objects
        end
    end
    local faceApplied = false
    if option.face and not faceIsAccessory then
        local ok, r = pcall(applyLocalFace, character, option.face)
        faceApplied = ok and r == true
    end
    local headlessApplied = false
    if option.headless then enforceHeadless(character, option); headlessApplied = true end
    local legApplied = false
    if option.hideLeftLeg then hideLeftLeg(character); legApplied = true end
    if option.hideRightLeg then hideRightLeg(character); legApplied = true end
    state.enabled = attached > 0 or faceApplied or headlessApplied or legApplied
    if state.enabled then
        if option.preserveRig and (option.hideLeftLeg or option.hideRightLeg) then enforceHeadless(character, option) end
        if option.hideOriginal then hideOriginalBody(character) end
        if option.hideOriginalHead then hideOriginalHead(character) end
        if option.head then hideOriginalHead(character) end
        if option.clearOriginalWearables then hideOriginalWearables(character, true) end
    end
    return state.enabled
end
local function restoreOriginalDescription()
    local description = state.originalDescription
    local character = LP.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if description and humanoid and character == state.descriptionCharacter and not state.skipDescriptionRestore then
        local clone = description:Clone()
        local ok = pcall(function() humanoid:ApplyDescription(clone) end)
        if not ok then pcall(function() humanoid:ApplyDescriptionReset(clone) end) end
        clone:Destroy()
    end
    if state.originalDescription then state.originalDescription:Destroy() end
    state.originalDescription = nil
    state.descriptionCharacter = nil
    state.skipDescriptionRestore = false
end
local function removeOutfit()
    state.enabled = false
    clearVisuals()
    restoreOriginalDescription()
end
return {
    outfits = OUTFITS,
    apply = function(index)
        state.selected = math.clamp(math.floor(tonumber(index) or 1), 1, #OUTFITS)
        return applyOutfit()
    end,
    remove = removeOutfit,
    cleanup = function()
        state.enabled = false
        clearVisuals()
        restoreOriginalDescription()
    end,
}

local function stopAutoSteal()
	disconnect("AutoSteal")
end
local function findStealPrompt(maxDist)
	local hrp = root()
	if not hrp then return nil, nil end
	local best, bestDist = nil, maxDist or 18
	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("ProximityPrompt") and obj.Enabled then
			local parent = obj.Parent
			local pos
			if parent and parent:IsA("BasePart") then
				pos = parent.Position
			elseif parent then
				local bp = parent:FindFirstChildWhichIsA("BasePart")
				pos = bp and bp.Position
			end
			if pos then
				local d = (pos - hrp.Position).Magnitude
				if d < bestDist then
					local action = (obj.ActionText or ""):lower()
					local objText = (obj.ObjectText or ""):lower()
					if action:find("steal") or action:find("grab") or action:find("take")
						or objText:find("steal") or objText:find("brain") or d < 10 then
						best, bestDist = obj, d
					end
				end
			end
		end
	end
	return best, bestDist
end
local function fireStealPrompt(prompt)
	if not prompt then return end
	pcall(function()
		if fireproximityprompt then
			fireproximityprompt(prompt)
		else
			prompt:InputHoldBegin()
			task.delay(0.05, function() pcall(function() prompt:InputHoldEnd() end) end)
		end
	end)
end
local function startAutoSteal()
	stopAutoSteal()
	Connections.AutoSteal = RunService.Heartbeat:Connect(function()
		if not Enabled.AutoSteal then return end
		local mode = Values.AutoStealMode or "Full"
		local prompt = findStealPrompt(mode == "Semi" and 12 or 18)
		if not prompt then return end
		if mode == "Semi" then
			local last = prompt:GetAttribute("_22S_SemiTick") or 0
			if tick() - last < 0.35 then return end
			prompt:SetAttribute("_22S_SemiTick", tick())
			fireStealPrompt(prompt)
			pcall(function()
				local gv = getgenv and getgenv() or _G
				gv._AS_StealProgress = math.clamp((tonumber(gv._AS_StealProgress) or 0) + 0.08, 0, 0.86)
			end)
		else
			fireStealPrompt(prompt)
		end
	end)
end

	stopAntiDie()
	if not hum then return end
	pcall(function()
		hum.BreakJointsOnDeath = false
		hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
	end)
	AntiDieState.healthConn = hum.HealthChanged:Connect(function(hp)
		if not Enabled.AntiDie then return end
		if hp <= 0 or tick() < AntiDieState.invincibleUntil then
			pcall(function()
				hum.Health = math.max(hum.Health, hum.MaxHealth * 0.35)
				hum:ChangeState(Enum.HumanoidStateType.Running)
			end)
			AntiDieState.invincibleUntil = tick() + 1.2
		end
	end)
	AntiDieState.diedConn = hum.Died:Connect(function()
		if not Enabled.AntiDie then return end
		pcall(function()
			hum.Health = hum.MaxHealth
			hum:ChangeState(Enum.HumanoidStateType.Running)
			hum.BreakJointsOnDeath = false
		end)
		AntiDieState.invincibleUntil = tick() + 1.5
	end)
end

	local hum = humanoid()

	disconnect("AntiDieChar")

		task.wait(0.3)
		local h = char:FindFirstChildOfClass("Humanoid")
		if h and Enabled.AntiDie then hookAntiDieHumanoid(h) end
	end)
end

-- Drop brainrot (ascend then snap to ground)
local dropActive = false
local DROP_ASCEND_DURATION = 0.35
local DROP_ASCEND_SPEED = 90
local function runDrop()
	if dropActive then return end
	local char = character()
	if not char then return end
	local r = char:FindFirstChild("HumanoidRootPart")
	if not r then return end
	dropActive = true
	local t0 = tick()
	local dc
	dc = RunService.Heartbeat:Connect(function()
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if not root then
			dc:Disconnect()
			dropActive = false
			return
		end
		if tick() - t0 >= DROP_ASCEND_DURATION then
			dc:Disconnect()
			local rp = RaycastParams.new()
			rp.FilterDescendantsInstances = {char}
			rp.FilterType = Enum.RaycastFilterType.Exclude
			local rr = workspace:Raycast(root.Position, Vector3.new(0, -2000, 0), rp)
			if rr then
				local hum2 = char:FindFirstChildOfClass("Humanoid")
				local off = (hum2 and hum2.HipHeight or 2) + (root.Size.Y / 2)
				root.CFrame = CFrame.new(root.Position.X, rr.Position.Y + off, root.Position.Z)
				root.AssemblyLinearVelocity = Vector3.zero
			end
			dropActive = false
			return
		end
		local v = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(v.X, DROP_ASCEND_SPEED, v.Z)
	end)
end

local function switchTab(name)
	currentTab = name
	for n, page in pairs(pages) do
		page.Visible = (n == name)
	end
	for n, btn in pairs(tabBtns) do
		local on = (n == name)
		TweenService:Create(btn, TweenInfo.new(0.18), {
			BackgroundColor3 = on and C.tabActive or C.tabBg,
			BackgroundTransparency = on and 0.05 or 0.2
		}):Play()
		local lbl = btn:FindFirstChild("TabLabel")
		if lbl then
			lbl.TextColor3 = on and C.text or C.textDim
		end
		local under = btn:FindFirstChild("Underline")
		if under then
			TweenService:Create(under, TweenInfo.new(0.2), {
				BackgroundTransparency = on and 0 or 1,
				Size = on and UDim2.new(0.6, 0, 0, 2 * guiScale) or UDim2.new(0, 0, 0, 2 * guiScale)
			}):Play()
		end
	end
end

	valueInput.ClearTextOnFocus = false
	valueInput.ZIndex = 8
	Instance.new("UICorner", valueInput).CornerRadius = UDim.new(0, 7 * guiScale)
	local boxStroke = Instance.new("UIStroke", valueInput)
	boxStroke.Color = C.purple
	boxStroke.Thickness = 1
	boxStroke.Transparency = 0.45

	local function setValue(val)
		val = math.clamp(math.floor(val), minVal, maxVal)
		valueInput.Text = tostring(val)
		Values[valueKey] = val
	end
	SliderSetters[valueKey] = setValue

	valueInput.FocusLost:Connect(function()
		local n = tonumber(valueInput.Text)
		if n then
			setValue(n)
			callback(Values[valueKey])
		else
			valueInput.Text = tostring(Values[valueKey])
		end
	end)
end

	sliderFill.ZIndex = 9
	Instance.new("UICorner", sliderFill).CornerRadius = UDim.new(1, 0)

	local thumb = Instance.new("Frame", sliderBg)
	thumb.Size = UDim2.new(0, 14 * guiScale, 0, 14 * guiScale)
	thumb.Position = UDim2.new(pct, -7 * guiScale, 0.5, -7 * guiScale)
	thumb.BackgroundColor3 = Color3.new(1, 1, 1)
	thumb.ZIndex = 10
	Instance.new("UICorner", thumb).CornerRadius = UDim.new(1, 0)

	local sliderBtn = Instance.new("TextButton", sliderBg)
	sliderBtn.Size = UDim2.new(1, 0, 3, 0)
	sliderBtn.Position = UDim2.new(0, 0, -1, 0)
	sliderBtn.BackgroundTransparency = 1
	sliderBtn.Text = ""
	sliderBtn.ZIndex = 11

	local dragging = false
	local function updateSlider(rel)
		rel = math.clamp(rel, 0, 1)
		sliderFill.Size = UDim2.new(rel, 0, 1, 0)
		thumb.Position = UDim2.new(rel, -7 * guiScale, 0.5, -7 * guiScale)
		local val = math.floor(minVal + (maxVal - minVal) * rel)
		valueInput.Text = tostring(val)
		Values[valueKey] = val
		callback(val)
	end

	local function setSliderValue(val)
		val = math.clamp(val, minVal, maxVal)
		local rel = (val - minVal) / (maxVal - minVal)
		sliderFill.Size = UDim2.new(rel, 0, 1, 0)
		thumb.Position = UDim2.new(rel, -7 * guiScale, 0.5, -7 * guiScale)
		valueInput.Text = tostring(val)
		Values[valueKey] = val
	end
	SliderSetters[valueKey] = setSliderValue

	sliderBtn.MouseButton1Down:Connect(function() dragging = true end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			updateSlider((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X)
		end
	end)

	valueInput.FocusLost:Connect(function()
		local n = tonumber(valueInput.Text)
		if n then
			n = math.clamp(math.floor(n), minVal, maxVal)
			setSliderValue(n)
			callback(n)
		else
			valueInput.Text = tostring(Values[valueKey])
		end
	end)
end

UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end

	if waitingForKeybind and input.KeyCode ~= Enum.KeyCode.Unknown then
		KEYBINDS[waitingForKeybind] = input.KeyCode
		if KeyButtons[waitingForKeybind] then
			KeyButtons[waitingForKeybind].Text = input.KeyCode.Name
		end
		waitingForKeybind = nil
		return
	end

	if input.KeyCode == KEYBINDS.HIDEMENU then
		guiVisible = not guiVisible
		main.Visible = guiVisible
		return
	end

	if input.KeyCode == KEYBINDS.SPIN then
		Enabled.SpinBot = not Enabled.SpinBot
		if VisualSetters.SpinBot then VisualSetters.SpinBot(Enabled.SpinBot) end
		if Enabled.SpinBot then startSpinBot() else stopSpinBot() end
	end

	if input.KeyCode == KEYBINDS.CARRYSWAP then
		SpeedMode.Carry = not SpeedMode.Carry
		refreshSpeedLabel()
	end

	if input.KeyCode == KEYBINDS.LAGGERSWAP then
		SpeedMode.Lagger = not SpeedMode.Lagger
		refreshSpeedLabel()
	end

	if input.KeyCode == KEYBINDS.TPDOWN then
		runTPFloor()
	end

	if input.KeyCode == KEYBINDS.AIMBOT then
		Enabled.BatAimbot = not Enabled.BatAimbot
		if VisualSetters.BatAimbot then VisualSetters.BatAimbot(Enabled.BatAimbot) end
		if Enabled.BatAimbot then startBatAimbot() else stopBatAimbot() end
	end

	if input.KeyCode == KEYBINDS.TPBAT then
		Enabled.TPBat = not Enabled.TPBat
		if VisualSetters.TPBat then VisualSetters.TPBat(Enabled.TPBat) end
		if Enabled.TPBat then startTPBat() else stopTPBat() end
	end

	if input.KeyCode == KEYBINDS.INSTANTRESET then
		runInstantReset()
	end

	if input.KeyCode == KEYBINDS.DROP then
		runDrop()
	end
end)

				dragging = true
				start = input.Position
				startPos = pad.Position
			end

			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
				local d = input.Position - start
				pad.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
			end
		end)
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
			end
		end)
	end

	local buttons = {}

	local function applyShape(btn)
		local corner = btn:FindFirstChildOfClass("UICorner")
		if not corner then
			corner = Instance.new("UICorner", btn)
		end
		if shapeMode == "circle" then
			corner.CornerRadius = UDim.new(1, 0)
		else
			corner.CornerRadius = UDim.new(0, 12)

		st.Transparency = 0.25
		applyShape(btn)

		local dragging, moved, start, startPos
		btn.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = true
				moved = false
				start = input.Position
				startPos = btn.Position
			end
		end)
		UserInputService.InputChanged:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
				local d = input.Position - start
				if d.Magnitude > 6 then moved = true end
				btn.Position = UDim2.new(0, startPos.X.Offset + d.X, 0, startPos.Y.Offset + d.Y)
			end
		end)
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				if dragging and not moved then

				dragging = false
			end
